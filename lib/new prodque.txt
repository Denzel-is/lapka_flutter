-- Удаляем таблицы, если они существуют (чтобы избежать ошибок)
DROP TABLE IF EXISTS cart_items CASCADE;
DROP TABLE IF EXISTS products CASCADE;
DROP TABLE IF EXISTS categories CASCADE;
DROP TABLE IF EXISTS users CASCADE;

-- Создаем таблицу пользователей
CREATE TABLE users (
  id SERIAL PRIMARY KEY,                -- Уникальный идентификатор пользователя (auto-increment)
  email VARCHAR(255) NOT NULL UNIQUE,   -- Email пользователя
  password VARCHAR(255) NOT NULL        -- Хэшированный пароль пользователя
);

-- Создаем таблицу категорий
CREATE TABLE categories (
  id SERIAL PRIMARY KEY,               -- Уникальный идентификатор категории (auto-increment)
  name VARCHAR(255) NOT NULL UNIQUE    -- Название категории
);

-- Создаем таблицу продуктов
CREATE TABLE products (
  id SERIAL PRIMARY KEY,               -- Уникальный идентификатор продукта (auto-increment)
  name VARCHAR(255) NOT NULL,          -- Название продукта
  price NUMERIC(10, 2) NOT NULL,       -- Цена продукта
  category_id INT NOT NULL,            -- Внешний ключ на таблицу категорий
  imageUrl TEXT,                       -- Ссылка на изображение продукта
  CONSTRAINT fk_category FOREIGN KEY (category_id) REFERENCES categories(id) ON DELETE CASCADE
);

-- Создаем таблицу элементов корзины
CREATE TABLE cart_items (
  id SERIAL PRIMARY KEY,               -- Уникальный идентификатор записи корзины (auto-increment)
  user_id INT NOT NULL,                -- Внешний ключ на таблицу пользователей
  product_id INT NOT NULL,             -- Внешний ключ на таблицу продуктов
  quantity INT NOT NULL,               -- Количество товаров
  CONSTRAINT fk_user FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE,
  CONSTRAINT fk_product FOREIGN KEY (product_id) REFERENCES products(id) ON DELETE CASCADE
);

-- Добавляем индексы для оптимизации
CREATE INDEX idx_products_name ON products (name);
CREATE INDEX idx_products_category_id ON products (category_id);
